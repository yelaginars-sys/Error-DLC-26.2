package dev.liteproject.client.ui;

import dev.liteproject.client.ClientCore;
import dev.liteproject.client.render.ui.UiDrawList;

/** Interactive letter-by-letter technical title beneath the CsClickGui panel. */
final class CsTechTitle {
    private static final String TITLE="Lite Project";
    private static final String GLITCH_CHARS="ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789";
    private static final int WHITE=0xFFFFFFFF;
    private final Glyph[] glyphs=new Glyph[TITLE.length()];
    private float lastSize,lastCenter,sweep,lensX,frameAlpha;
    private int grabbed=-1;
    private float grabX,grabY;

    CsTechTitle(){for(int i=0;i<glyphs.length;i++)glyphs[i]=new Glyph(TITLE.substring(i,i+1),i);}

    void draw(UiDrawList ui,float centerX,float centerY,float size,float scale,
              int mx,int my,boolean down,boolean pressed,float dt){
        if(size<=0)return;
        layout(centerX,size);
        float left=glyphs[0].x,right=glyphs[glyphs.length-1].x+glyphs[glyphs.length-1].width;
        boolean inside=mx>=left-15*scale&&mx<=right+15*scale
            &&my>=centerY-size*.65f-17*scale&&my<=centerY+size*.65f+17*scale;
        if(pressed&&inside){
            int focus=nearest(mx);
            if(focus>=0){grabbed=focus;grabX=mx-glyphs[focus].dx;grabY=my-glyphs[focus].dy;}
        }
        if(!down)grabbed=-1;
        if(grabbed>=0){
            Glyph glyph=glyphs[grabbed];
            glyph.dx+=(mx-grabX-glyph.dx)*(1-(float)Math.exp(-dt/.03f));
            glyph.dy+=(my-grabY-glyph.dy)*(1-(float)Math.exp(-dt/.03f));
            glyph.vx=glyph.vy=0;
        }
        for(int i=0;i<glyphs.length;i++){
            if(i==grabbed)continue;
            Glyph g=glyphs[i];
            g.vx+=(-320*g.dx-22*g.vx)*dt;
            g.vy+=(-320*g.dy-22*g.vy)*dt;
            g.dx+=g.vx*dt;g.dy+=g.vy*dt;
            if(Math.abs(g.dx)+Math.abs(g.dy)+Math.abs(g.vx)+Math.abs(g.vy)<.08f){g.dx=g.dy=g.vx=g.vy=0;}
        }
        sweep+=dt;
        float target=inside?mx:left+(right-left)*(.5f-.5f*(float)Math.cos(sweep*.45f));
        lensX+=(target-lensX)*(1-(float)Math.exp(-dt/(inside?.05f:.22f)));
        int focus=grabbed>=0?grabbed:nearest(lensX);
        frameAlpha+=(1-frameAlpha)*(1-(float)Math.exp(-dt/.1f));
        for(int i=0;i<glyphs.length;i++){
            Glyph g=glyphs[i];
            if(g.text.isBlank())continue;
            g.display=scrambled(g,sweep);
            g.drawnWidth=g.display.equals(g.text)?g.width:ClientCore.getInstance().renderer().textWidth(g.display,size,true);
            float drawX=g.x+g.dx+(g.width-g.drawnWidth)*.5f;
            boolean active=i==focus;
            g.highlight+=((active?1:0)-g.highlight)*(1-(float)Math.exp(-dt/.09f));
            int base=withAlpha(WHITE,1-.53f*g.highlight);
            if(Math.abs(g.dx)+Math.abs(g.dy)>1){
                ui.strongText(g.x,centerY,size,0x55FFFFFF,g.text);
                ui.line(g.x+g.width*.5f,centerY,g.x+g.width*.5f+g.dx,centerY+g.dy,
                    Math.max(.5f,scale),0x77FFFFFF);
            }
            ui.strongText(drawX,centerY+g.dy,size,base,g.display);
            if(g.highlight>.02f){
                dashedGlyph(ui,g.display,drawX,g.drawnWidth,centerY+g.dy,size,scale,g.highlight);
                float a=g.highlight;
                // A dotted light trace follows the contour area of the selected glyph.
                float bx=g.x+g.dx-2*scale,by=centerY+g.dy-size*.54f;
                float bw=g.width+4*scale,bh=size*1.08f;
                dashedBox(ui,bx,by,bw,bh,4*scale,2*scale,Math.max(.55f,scale*.9f),withAlpha(WHITE,a*.6f));
            }
        }
        if(focus>=0&&frameAlpha>.01f)drawFrame(ui,glyphs[focus],centerY,size,scale,frameAlpha,sweep);
    }

    private void layout(float center,float size){
        if(size==lastSize&&center==lastCenter)return;
        lastSize=size;lastCenter=center;
        float spacing=-.05f*size,width=0;
        for(Glyph glyph:glyphs){glyph.width=glyph.text.isBlank()?size*.33f:ClientCore.getInstance().renderer().textWidth(glyph.text,size,true);width+=glyph.width;}
        width+=spacing*(glyphs.length-1);
        float x=center-width*.5f;
        for(Glyph glyph:glyphs){glyph.x=x;x+=glyph.width+spacing;}
        lensX=center;
    }

    private int nearest(float x){
        int best=-1;float distance=Float.MAX_VALUE;
        for(int i=0;i<glyphs.length;i++){
            Glyph g=glyphs[i];if(g.text.isBlank())continue;
            float lo=g.x+g.dx,hi=lo+g.width;
            float d=x<lo?lo-x:x>hi?x-hi:0;
            if(d<distance){distance=d;best=i;}
        }
        return best;
    }

    private static void dashedGlyph(UiDrawList ui,String glyph,float x,float width,float y,float size,float scale,float alpha){
        float dash=Math.max(.7f,4*scale),gap=Math.max(.4f,2*scale);
        float offset=Math.max(.5f,1.2f*scale);
        int color=withAlpha(WHITE,alpha*.95f);
        for(float slice=0;slice<width;slice+=dash+gap){
            ui.pushScissor(x+slice-offset,y-size*.75f,Math.min(dash,width-slice)+offset*2,size*1.5f);
            ui.strongText(x-offset,y,size,color,glyph).strongText(x+offset,y,size,color,glyph)
              .strongText(x,y-offset,size,color,glyph).strongText(x,y+offset,size,color,glyph);
            ui.popScissor();
        }
        ui.strongText(x,y,size,withAlpha(0xFF0B0B0C,alpha),glyph);
    }
    private static void drawFrame(UiDrawList ui,Glyph g,float cy,float size,float scale,float alpha,float time){
        float x=g.x+g.dx-6*scale,y=cy+g.dy-size*.6f-6*scale;
        float w=Math.max(g.width,g.drawnWidth)+12*scale,h=size*1.2f+12*scale;
        int c=withAlpha(WHITE,.4f*alpha);
        ui.outlineRoundedRect(x,y,w,h,0,Math.max(.5f,scale*.8f),c);
        float corner=2*scale;
        for(float[] p:new float[][]{{x,y},{x+w,y},{x+w,y+h},{x,y+h}})
            ui.rect(p[0]-corner,p[1]-corner,corner*2,corner*2,withAlpha(WHITE,.86f*alpha));
        int seed=g.index+1;
        float perimeter=2*(w+h);
        for(int k=0;k<15;k++){
            float phase=(time/(.5f+noise(seed,k,11)*1.2f)+noise(seed,k,17));
            float life=phase-(float)Math.floor(phase);
            if(life>.7f)continue;
            float[] p=point(x,y,w,h,noise(seed,k,(int)phase)*perimeter);
            float outward=(3+noise(seed,k,29)*8)*scale;
            float px=p[0]+p[2]*outward,py=p[1]+p[3]*outward;
            float side=(k%5==0?2:1)*scale;
            ui.rect(px,py,Math.max(.5f,side),Math.max(.5f,side),withAlpha(WHITE,alpha*(1-life)*.62f));
        }
        float[] pulse=point(x,y,w,h,(time*.42f*perimeter)%perimeter);
        ui.rect(pulse[0],pulse[1],Math.max(1,2*scale),Math.max(1,2*scale),withAlpha(WHITE,alpha*.9f));
        ui.text(x,y-8*scale,7*scale,withAlpha(WHITE,alpha*.7f),
            g.display+"  "+Math.round(g.drawnWidth)+" × "+Math.round(size));
    }

    private static String scrambled(Glyph glyph,float time){
        float cycleLength=3.2f;
        int cycle=(int)(time/cycleLength);
        float phase=time-cycle*cycleLength;
        if(phase>.85f)return glyph.text;
        int count=1+(int)(noise(cycle,37,11)*5);
        float score=noise(cycle,glyph.index,42);
        int rank=0;
        for(int i=0;i<TITLE.length();i++){
            if(TITLE.charAt(i)==' '||i==glyph.index)continue;
            float other=noise(cycle,i,42);
            if(other<score||(other==score&&i<glyph.index))rank++;
        }
        if(rank>=count)return glyph.text;
        int frame=(int)(phase/.09f);
        int chosen=Math.min(GLITCH_CHARS.length()-1,(int)(noise(cycle*67+frame,glyph.index,91)*GLITCH_CHARS.length()));
        return GLITCH_CHARS.substring(chosen,chosen+1);
    }
    private static float[] point(float x,float y,float w,float h,float d){
        if(d<w)return new float[]{x+d,y,0,-1};d-=w;
        if(d<h)return new float[]{x+w,y+d,1,0};d-=h;
        if(d<w)return new float[]{x+w-d,y+h,0,1};d-=w;
        return new float[]{x,y+h-d,-1,0};
    }
    private static void dashedBox(UiDrawList ui,float x,float y,float w,float h,float dash,float gap,float stroke,int color){
        for(float dx=0;dx<w;dx+=dash+gap){float end=Math.min(w,dx+dash);ui.line(x+dx,y,x+end,y,stroke,color).line(x+dx,y+h,x+end,y+h,stroke,color);}
        for(float dy=0;dy<h;dy+=dash+gap){float end=Math.min(h,dy+dash);ui.line(x,y+dy,x,y+end,stroke,color).line(x+w,y+dy,x+w,y+end,stroke,color);}
    }
    private static float noise(int a,int b,int c){int h=a*73856093^b*19349663^c*83492791;h^=h>>>16;return (h&0x7FFFFFFF)/(float)Integer.MAX_VALUE;}
    private static int withAlpha(int color,float alpha){return (color&0xFFFFFF)|(Math.round(Math.max(0,Math.min(1,alpha))*255)<<24);}
    private static final class Glyph{
        final String text;final int index;String display;float x,width,drawnWidth,dx,dy,vx,vy,highlight;
        Glyph(String text,int index){this.text=text;this.display=text;this.index=index;}
    }
}
